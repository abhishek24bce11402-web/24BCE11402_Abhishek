-- 1. List the Employee Name, their Department Name, and the Location Name for all employees.
SELECT e.employee_name, d.dept_name, d.location_id
FROM Employees e
JOIN Departments d
ON e.dept_id = d.dept_id;



-- 2. Retrieve the Project Name, Client Name, and the Total Hours Worked on that project,
-- only for projects managed by the 'Consulting' department.
SELECT p.project_name, p.client_name, SUM(a.hours_worked) AS total_hours
FROM Projects p
JOIN Assignments a ON p.project_id = a.project_id
JOIN Departments d ON p.dept_id = d.dept_id
WHERE d.dept_name = 'Consulting'
GROUP BY p.project_name, p.client_name;



-- 3. Find the names of Employees who have worked on a project with Client Name 'GlobalTech Solutions'.
SELECT e.employee_name, p.project_name
FROM Employees e
JOIN Assignments a ON e.employee_id = a.employee_id
JOIN Projects p ON a.project_id = p.project_id
WHERE p.client_name = 'GlobalTech Solutions';




-- 5. Show the Manager Name, the Department Name they manage,
-- and the total number of direct reports they have.
SELECT m.employee_name AS manager_name,
       d.dept_name,
       COUNT(e.employee_id) AS direct_reports
FROM Employees e
JOIN Employees m
ON e.manager_id = m.employee_id
JOIN Departments d
ON m.dept_id = d.dept_id
GROUP BY m.employee_name, d.dept_name;



-- 6. Calculate the Average Hours Worked per Employee across all projects.
SELECT e.employee_name,
       AVG(a.hours_worked) AS avg_hours
FROM Employees e
JOIN Assignments a
ON e.employee_id = a.employee_id
GROUP BY e.employee_name;



-- 7. Find the Department Name(s) that have an average salary above the company-wide average salary.
SELECT d.dept_name
FROM Employees e
JOIN Departments d
ON e.dept_id = d.dept_id
GROUP BY d.dept_name
HAVING AVG(e.salary) >
(
    SELECT AVG(salary)
    FROM Employees
);



-- 8. Identify the Client Name(s) whose projects have a total combined hours worked exceeding 500 hours.
SELECT p.client_name,
       SUM(a.hours_worked) AS total_hours
FROM Projects p
JOIN Assignments a
ON p.project_id = a.project_id
GROUP BY p.client_name
HAVING SUM(a.hours_worked) > 500;



-- 9. For each Location, find the Department Name with the highest count of employees.
SELECT d.location_id,
       d.dept_name,
       COUNT(e.employee_id) AS total_employees
FROM Departments d
JOIN Employees e
ON d.dept_id = e.dept_id
GROUP BY d.location_id, d.dept_name
HAVING COUNT(e.employee_id) =
(
    SELECT MAX(emp_count)
    FROM
    (
        SELECT COUNT(e2.employee_id) AS emp_count
        FROM Departments d2
        JOIN Employees e2
        ON d2.dept_id = e2.dept_id
        WHERE d2.location_id = d.location_id
        GROUP BY d2.dept_name
    ) temp
);



-- 10. Calculate the total number of employees assigned to each Project,
-- only include projects that have more than 5 employees assigned.
SELECT p.project_name,
       COUNT(DISTINCT a.employee_id) AS total_employees
FROM Projects p
JOIN Assignments a
ON p.project_id = a.project_id
GROUP BY p.project_name
HAVING COUNT(DISTINCT a.employee_id) > 5;



-- 11. Find the Employee Name(s) who work on ALL projects managed by the 'R&D' department.
SELECT e.employee_name
FROM Employees e
WHERE NOT EXISTS
(
    SELECT p.project_id
    FROM Projects p
    JOIN Departments d
    ON p.dept_id = d.dept_id
    WHERE d.dept_name = 'R&D'
    AND NOT EXISTS
    (
        SELECT *
        FROM Assignments a
        WHERE a.employee_id = e.employee_id
        AND a.project_id = p.project_id
    )
);



-- 12. Retrieve the Department Name(s) that have employees assigned to more than one project.
SELECT d.dept_name
FROM Departments d
JOIN Employees e
ON d.dept_id = e.dept_id
JOIN Assignments a
ON e.employee_id = a.employee_id
GROUP BY d.dept_name
HAVING COUNT(DISTINCT a.project_id) > 1;



-- 13. List the Employee Name and Project Name for any employee
-- who has worked less than 10 hours on a project.
SELECT e.employee_name,
       p.project_name
FROM Employees e
JOIN Assignments a
ON e.employee_id = a.employee_id
JOIN Projects p
ON a.project_id = p.project_id
WHERE a.hours_worked < 10;



-- 15. Select the Employee Name(s) whose salary is greater than the maximum salary in their own department.
SELECT e.employee_name
FROM Employees e
WHERE e.salary >
(
    SELECT MAX(salary)
    FROM Employees
    WHERE dept_id = e.dept_id
    AND employee_id <> e.employee_id
);





-- 19. Find the top 3 Departments based on the total hours worked by all their employees across all projects.
SELECT d.dept_name,
       SUM(a.hours_worked) AS total_hours
FROM Departments d
JOIN Employees e
ON d.dept_id = e.dept_id
JOIN Assignments a
ON e.employee_id = a.employee_id
GROUP BY d.dept_name
ORDER BY total_hours DESC
LIMIT 3;



-- 20. Display the Project Name and a status
-- ("High Activity" if total hours > 300, "Normal" otherwise).
SELECT p.project_name,
       CASE
           WHEN SUM(a.hours_worked) > 300 THEN 'High Activity'
           ELSE 'Normal'
       END AS status
FROM Projects p
LEFT JOIN Assignments a
ON p.project_id = a.project_id
GROUP BY p.project_name;



-- 21. List the Employee Name and their Manager's Name,
-- even if the employee has no manager.
SELECT e.employee_name AS employee_name,
       IFNULL(m.employee_name, 'No Manager') AS manager_name
FROM Employees e
LEFT JOIN Employees m
ON e.manager_id = m.employee_id;



-- 22. Use a CTE to find the Project Name and the number of unique employees assigned to it.
WITH ProjectEmployeeCount AS
(
    SELECT p.project_name,
           COUNT(DISTINCT a.employee_id) AS employee_count
    FROM Projects p
    LEFT JOIN Assignments a
    ON p.project_id = a.project_id
    GROUP BY p.project_name
)
SELECT *
FROM ProjectEmployeeCount;



-- 23. Find the Employee Name(s) who are assigned to the same projects as Employee ID 101.
SELECT DISTINCT e.employee_name
FROM Employees e
JOIN Assignments a
ON e.employee_id = a.employee_id
WHERE a.project_id IN
(
    SELECT project_id
    FROM Assignments
    WHERE employee_id = 101
)
AND e.employee_id <> 101;
